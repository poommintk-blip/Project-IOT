/*
* Smart Weather Lamp & Digital Twin Firmware
* Course: DGT01 1930 IoT Project
* Target Board: ESP32 Development Board
*/

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>

// Pin Definitions
#define DHTPIN       4       // DHT22 Data Pin
#define DHTTYPE      DHT22   // DHT22 Sensor Type
#define LDR_PIN      34      // LDR Photoresistor (Analog ADC Pin)

#define LED_BLUE     16      // Status LED Blue (Rain)
#define LED_YELLOW   17      // Status LED Yellow (Normal Sun)
#define LED_RED      18      // Status LED Red (Intense Sun / Hot)

// Initialize Peripherals
DHT dht(DHTPIN, DHTTYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2); // Address 0x27 or 0x3F

void setup() {
  Serial.begin(115200);
  
  // Pin Configurations
  pinMode(LED_BLUE, OUTPUT);
  pinMode(LED_YELLOW, OUTPUT);
  pinMode(LED_RED, OUTPUT);
  
  // Digital Outputs LOW initially
  digitalWrite(LED_BLUE, LOW);
  digitalWrite(LED_YELLOW, LOW);
  digitalWrite(LED_RED, LOW);

  // Initialize Sensors & LCD
  dht.begin();
  Wire.begin(21, 22); // SDA=21, SCL=22
  lcd.init();
  lcd.backlight();

  // Startup Screen
  lcd.setCursor(0, 0);
  lcd.print("Smart Weather");
  lcd.setCursor(0, 1);
  lcd.print("Digital Twin init");
  delay(2000);
  lcd.clear();
}

void loop() {
  // Read Sensors
  float humidity = dht.readHumidity();
  float temperature = dht.readTemperature();
  int ldrValue = analogRead(LDR_PIN); // 0 - 4095 on ESP32

  // Fallback for Sensor Errors
  if (isnan(humidity) || isnan(temperature)) {
    humidity = 50.0;
    temperature = 25.0;
  }

  // Weather Logic Decision Branching
  String weatherStatus = "NORMAL";

  // Logic 1: High Humidity (> 85%) -> RAIN
  if (humidity >= 80.0) {
    weatherStatus = "RAIN";
    digitalWrite(LED_BLUE, HIGH);
    digitalWrite(LED_YELLOW, LOW);
    digitalWrite(LED_RED, LOW);
  }
  // Logic 2: Intense Sunlight (LDR ADC > 2500) -> HOT / INTENSE SUN
  else if (ldrValue > 2500) {
    weatherStatus = "HOT";
    digitalWrite(LED_BLUE, LOW);
    digitalWrite(LED_YELLOW, LOW);
    digitalWrite(LED_RED, HIGH);
  }
  // Logic 3: Normal Conditions -> NORMAL SUN
  else {
    weatherStatus = "NORMAL";
    digitalWrite(LED_BLUE, LOW);
    digitalWrite(LED_YELLOW, HIGH);
    digitalWrite(LED_RED, LOW);
  }

  // Update Physical LCD Screen
  lcd.setCursor(0, 0);
  lcd.print("T:");
  lcd.print(temperature, 1);
  lcd.print((char)223); // Degree symbol
  lcd.print("C H:");
  lcd.print(humidity, 0);
  lcd.print("%  ");

  lcd.setCursor(0, 1);
  lcd.print("ST: ");
  lcd.print(weatherStatus);
  lcd.print("     ");

  // Output JSON Stream over Serial Port for Digital Twin Sync
  Serial.print("{\"temp\":");
  Serial.print(temperature, 1);
  Serial.print(",\"hum\":");
  Serial.print(humidity, 1);
  Serial.print(",\"ldr\":");
  Serial.print(ldrValue);
  Serial.print(",\"status\":\"");
  Serial.print(weatherStatus);
  Serial.println("\"}");

  delay(1000); // Send Telemetry Every 1 Second
}
